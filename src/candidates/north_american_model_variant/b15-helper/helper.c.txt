#include "../../types.h"
#include "model_variant.h"

typedef union {
    PSXLONG packed;
    DVECTOR point;
} B15Screen;

typedef struct {
    SVECTOR p[9];
    B15Screen s[9];
    s32 angle[9];
    SVECTOR e[9];
    B15Screen es[9];
    s32 width[9];
    u8 color[4];
    u8 pad120[0x4];
    u8 pad128[0x70];
    s32 depth[9];
    s16 xo[9];
    s16 yo[9];
    s32 amount[9];
    u8 pad204[0x4];
} B15Ring;

void func_8013CE7C(u8 *ctx)
{
    SVECTOR rot;
    VECTOR scale;
    MATRIX m;
    MATRIX ls;
    GsCOORDINATE2 coord;
    PSXLONG flags[16][9];
    PSXLONG p;
    PSXLONG flag;
    u8 *work;
    s16 i, j;
    GsOT *ot;
    s32 yaw, pitch;
    s32 a, b, lim;
    s32 size;
    s16 ext;
    s32 ang1, ang2, phase;
    s32 dx, dy;
    B15Ring *ring;
    POLY_FT4 *quad;
    s32 half;
    s32 sn;
    s32 off;

    SVECTOR *pe;
    u8 *sp4;
    u8 *sp2;
    s32 yv;

    work = ctx;
    ot = func_80058F10();
    yaw = ratan2(MODEL_VARIANT_WORD(work, 0x4298), MODEL_VARIANT_WORD(work, 0x4290)) + 0xC00;
    pitch = ratan2(MODEL_VARIANT_WORD(work, 0x4294), MODEL_VARIANT_WORD(work, 0x4290)) + 0xC00;
    if (!(MODEL_VARIANT_WORD(work, 0x42A8) & 1)) {
        size = MODEL_VARIANT_HALF(work, 0x42CE);
        if (size < 0) {
            size += 0x7F;
        }
        size = (u32)size >> 7;
    } else {
        size = MODEL_VARIANT_HALF(work, 0x42CE) * 12;
        if (size < 0) {
            size += 0x3FF;
        }
        size = (u32)size >> 10;

    i = 0;
    ang2 = 0;
    ext = (s16)size;
    lim = 0x400;

        for (j = 0; j < 9; j++, phase += 0x514) {
            a = ring->amount[j];
            b = 0x400;
            if (a > 0) {
                b = 0;
                if (a < 0x400) {
                    b = lim - a;
                }
            }
            ring->p[j].vx = (rcos(ang1) * (half = b / 2) >> 12) + (rcos(pitch + yaw + ang1) * (rsin(phase) << 5 >> 12) >> 12);

            pe = (SVECTOR *)((u8 *)pp + 0x90);
            pe->vx = pp->vx + (rcos(yaw) * ext >> 12);
            pe->vy = pp->vy;
            pe->vz = pp->vz + (rsin(yaw) * ext >> 12);
            a = ring->amount[j];
            if (a <= 0x400) {
                ring->amount[j] += MODEL_VARIANT_WORD(work, 0x42B4) << 7;
                if (ring->amount[j] >= 0x400) {
                    ring->amount[j] = 0x400;
                    if ((u32)MODEL_VARIANT_WORD(MODEL_VARIANT_WORD(work, 0x42BC), 0x20) < (u32)MODEL_VARIANT_WORD(work, 0x42AC)) {
                        if (j == 8) {
                            *(s32 *)((u8 *)ring + 0x204) = 1;
                        }
                    } else if (j == 8) {
                        for (j = 0; j < 9; j++) {
                            ring->amount[j] = -(j << 7);
                        }
                    }
                }
            }
        }
    }

    rot.vy = 0;
    rot.vz = 0;

    m.t[2] = MODEL_VARIANT_WORD(work, 0x4270);

    scale.vy = 0x1000;
    scale.vz = 0x1000;

    ScaleMatrix(&m, &scale);
    coord.coord = m;

    coord.flg = 0;
    GsGetLs(&coord, &ls);

                func_800878E0(&ring->e[8], &ring->es[8].packed, &p, &flag);
                dx = ring->s[8].point.vx - ring->s[7].point.vx;
                dy = ring->s[8].point.vy - ring->s[7].point.vy;
                ring->angle[8] = ratan2(dy, dx) - 0x400;
                ring->width[8] = ring->es[8].point.vx - ring->s[8].point.vx;
                ring->xo[8] = rcos(ring->angle[8]) * ring->width[8] >> 12;
                ring->yo[8] = rsin(ring->angle[8]) * ring->width[8] >> 12;
            } else {
                ring->depth[j] = RotTransPers4(&ring->p[j], &ring->p[j + 1], &ring->p[j], &ring->p[j + 1],

                dx = (&ring->s[j])[1].point.vx - ring->s[j].point.vx;
                dy = (&ring->s[j])[1].point.vy - ring->s[j].point.vy;
                ring->angle[j] = ratan2(dy, dx) - 0x400;
                ring->width[j] = ring->es[j].point.vx - ring->s[j].point.vx;
                ring->xo[j] = rcos(ring->angle[j]) * ring->width[j] >> 12;
                ring->yo[j] = rsin(ring->angle[j]) * ring->width[j] >> 12;
            }
        }
    }
    ring = (B15Ring *)(work + 0x1518);
    for (i = 0; i < 16; i++, ring++) {
        for (j = 0; j < 8; j++) {

            sp4 = (u8 *)(j * 4 + (s32)ring);
            sp2 = (u8 *)(j * 2 + (s32)ring);
            quad->x0 = *(s16 *)(sp4 + 0x48) + *(s16 *)(sp2 + 0x1BC);
            yv = *(s16 *)(sp4 + 0x4A);
            quad->y0 = *(s16 *)(sp2 + 0x1CE) + yv;
            quad->x1 = *(s16 *)(sp4 + 0x4C) + *(s16 *)(sp2 + 0x1BE);
            yv = *(s16 *)(sp4 + 0x4E);
            quad->y1 = *(s16 *)(sp2 + 0x1D0) + yv;
            quad->x2 = *(s16 *)(sp4 + 0x48) - *(s16 *)(sp2 + 0x1BC);
